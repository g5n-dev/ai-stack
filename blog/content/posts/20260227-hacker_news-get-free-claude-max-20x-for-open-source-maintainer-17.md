```python
# 示例1：GitHub仓库贡献者统计
def count_contributors(repo_url):
    """
    统计GitHub仓库的贡献者数量
    :param repo_url: 仓库URL (如 "https://github.com/user/repo")
    :return: 贡献者数量
    """
    from github import Github
    import re
    
    # 从URL提取用户名和仓库名
    match = re.search(r'github\.com/([^/]+)/([^/]+)', repo_url)
    if not match:
        return "无效的GitHub仓库URL"
    
    username, repo_name = match.groups()
    
    try:
        # 使用GitHub API获取贡献者
        g = Github()
        repo = g.get_repo(f"{username}/{repo_name}")
        contributors = repo.get_contributors()
        
        return f"仓库 {repo_name} 有 {contributors.totalCount} 位贡献者"
    except Exception as e:
        return f"获取失败: {str(e)}"

# 使用示例
print(count_contributors("https://github.com/torvalds/linux"))
```

```python
# 示例3：检测开源项目健康度
def check_repo_health(repo_url):
    """
    检查GitHub仓库的健康度指标
    :param repo_url: 仓库URL
    :return: 健康度报告
    """
    from github import Github
    import re
    
    match = re.search(r'github\.com/([^/]+)/([^/]+)', repo_url)
    if not match:
        return "无效的URL"
    
    username, repo_name = match.groups()
    
    try:
        g = Github()
        repo = g.get_repo(f"{username}/{repo_name}")
        
        # 获取关键指标
        stars = repo.stargazers_count
        open_issues = repo.open_issues_count
        forks = repo.forks_count
        last_commit = repo.get_commits()[0].commit.author.date
        
        # 计算健康度分数
        score = min(100, (stars * 2) + (forks) - (open_issues * 5))
        
        return {
            "健康度分数": score,
            "星标数": stars,
            "分支数": forks,
            "未关闭问题": open_issues,
            "最后提交": last_commit.strftime("%Y-%m-%d")
        }
    except Exception as e:
        return f"检查失败: {str(e)}"

# 使用示例
print(check_repo_health("https://github.com/python/cpython"))
```
