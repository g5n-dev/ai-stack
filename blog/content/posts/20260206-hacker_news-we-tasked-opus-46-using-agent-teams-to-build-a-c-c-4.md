```python
# 示例1：词法分析器
# 将源代码字符串分解为有意义的标记

import re

class Lexer:
    """简单的词法分析器，用于将C代码转换为Token流"""
    
    # 定义C语言的Token类型正则表达式
    token_specification = [
        ('NUMBER',   r'\d+(\.\d*)?'),  # 整数或小数
        ('ASSIGN',   r'='),            # 赋值运算符 =
        ('END',      r';'),            # 语句结束符 ;
        ('ID',       r'[A-Za-z_]\w*'), # 标识符/变量名
        ('OP',       r'[+\-*/]'),      # 算术运算符
        ('WHITESPACE', r'[ \t]+'),     # 空格和制表符(忽略)
        ('MISMATCH', r'.'),            # 其他不匹配的字符
    ]
    
    def __init__(self):
        # 编译正则表达式
        self.tok_regex = '|'.join('(?P<%s>%s)' % pair for pair in self.token_specification)
        self.re_prog = re.compile(self.tok_regex)
    
    def tokenize(self, code):
        """输入C代码字符串，返回Token列表"""
        tokens = []
        for mo in self.re_prog.finditer(code):
            kind = mo.lastgroup
            value = mo.group()
            if kind == 'WHITESPACE':
                continue  # 跳过空白字符
            elif kind == 'MISMATCH':
                raise RuntimeError(f'意外的字符: {value}')
            tokens.append((kind, value))
        return tokens

# 测试代码
lexer = Lexer()
code = "int a = 10 + 5;"
print(f"源代码: {code}")
print(f"Token流: {lexer.tokenize(code)}")
```

```python
# 示例2：抽象语法树 (AST) 定义
# 将Token流转换为树状结构以便后续处理

from dataclasses import dataclass
from typing import List, Union

@dataclass
class Number:
    """表示数字节点"""
    value: float

@dataclass
class Variable:
    """表示变量节点"""
    name: str

@dataclass
class BinOp:
    """表示二元运算节点 (如加法、减法)"""
    left: Union['Number', 'Variable']  # 左操作数
    op: str                            # 运算符 (+, -, *, /)
    right: Union['Number', 'Variable'] # 右操作数

@dataclass
class Assignment:
    """表示赋值语句节点"""
    name: str                          # 变量名
    expr: Union[Number, Variable, BinOp] # 赋值的表达式

# 示例：构建 AST 表示 "result = 10 + 5"
# 这对应于代码: result = 10 + 5;
ast_example = Assignment(
    name="result",
    expr=BinOp(
        left=Number(10),
        op="+",
        right=Number(5)
    )
)

print(f"AST 结构: 变量 '{ast_example.name}' 被赋值为 {ast_example.expr.left.value} {ast_example.expr.op} {ast_example.expr.right.value}")
```

# 示例3：简单的栈式虚拟机解释器
# 模拟编译器后端，执行编译后的指令

import operator

class StackVM:
    """
    一个基于栈的虚拟机，用于模拟执行编译器生成的指令。
    类似于Java虚拟机或Python解释器的简化原理。
    """
    def __init__(self):
        self.stack = []  # 数据栈
        # 定义指令对应的操作
        self.ops = {
            '+': operator.add,
            '-': operator.sub,
            '*': operator.mul,
            '/': operator.truediv,
        }
    
    def execute(self, instructions):
        """
        执行指令列表。
        指令格式示例: ['PUSH', 10], ['PUSH', 5], ['ADD'], ['POP']
        """
        for instr in instructions:
            if instr == 'POP':
                if self.stack:
                    return self.stack.pop()
            elif isinstance(instr, str) and instr in self.ops:
                # 如果是运算符，从栈顶弹出两个元素进行计算
                if len(self.stack) < 2:
                    raise ValueError("栈下溢：操作数不足")
                right = self.stack.pop()
                left = self.stack.pop()
                result = self.ops[instr](left, right)
                self.stack.append(result)
            else:
                # 默认为数值，压入栈中
                self.stack.append(instr)
        
        return self.stack[-1] if self.stack else None

## 最佳实践指南

### 实践 1：构建多智能体协作架构

**说明**:
在构建复杂系统（如 C 编译器）时，单一 AI 模型难以处理所有细节。通过建立专门的智能体团队，将不同职责分配给不同的智能体（如词法分析专家、语法分析专家、代码生成专家等），可以显著提高系统的模块化程度和开发效率。

**实施步骤**:
1. 定义项目所需的各个子模块和职责边界
2. 为每个模块创建专门的智能体角色
3. 建立智能体之间的通信协议和数据流规范
4. 设计中央协调器来管理智能体间的交互

**注意事项**:
- 确保各智能体的接口定义清晰且稳定
- 避免智能体职责重叠导致冲突
- 建立明确的决策机制以处理智能体间的分歧

---

### 实践 2：建立增量式开发与验证循环

**说明**:
编译器开发需要高度的准确性。采用增量式开发策略，从最小的可用子集开始（如支持基本变量声明的子语言），逐步扩展功能，并在每个阶段进行严格的测试验证，可以及早发现并修复错误。

**实施步骤**:
1. 定义语言的最小子集规范
2. 实现该子集的编译功能
3. 编写测试用例验证生成的汇编代码
4. 逐步添加新功能并重复测试验证

**注意事项**:
- 不要试图一次性实现所有功能
- 保持每个增量版本都是可运行的
- 为每个新增功能编写回归测试

---

### 实践 3：实施严格的测试驱动开发

**说明**:
在 AI 辅助编码中，代码生成可能包含隐藏错误。采用测试驱动开发（TDD）方法，先编写测试用例，再生成代码实现，可以确保编译器行为的正确性，并提供自动化的回归保护。

**实施步骤**:
1. 为每个语言特性编写测试用例
2. 运行测试确认失败
3. 使用 AI 生成实现代码
4. 运行测试验证实现正确性

**注意事项**:
- 测试用例应覆盖正常情况和边界条件
- 包含标准编译器测试套件（如 gcc 测试用例的子集）
- 验证生成的汇编代码的正确性

---

### 实践 4：利用现有规范和参考实现

**说明**:
C 语言标准（ISO/IEC 9899）提供了完整的语言规范。在开发过程中，严格参考标准文档，并研究现有开源编译器（如 GCC、Clang）的实现思路，可以避免重新发明轮子，确保合规性。

**实施步骤**:
1. 获取并熟悉 C 语言标准文档
2. 研究参考编译器的源代码结构
3. 对比参考实现与自身实现的差异
4. 使用标准测试套件验证合规性

**注意事项**:
- 注意版权问题，避免直接复制代码
- 理解规范背后的原理而非机械模仿
- 关注标准的版本差异（C89, C99, C11 等）

---

### 实践 5：设计清晰的中间表示（IR）

**说明**:
中间表示是编译器的核心，连接前端解析和后端代码生成。设计一个结构清晰、易于优化的 IR，可以简化编译器架构，使各智能体能够专注于特定阶段的处理。

**实施步骤**:
1. 设计 IR 的数据结构表示
2. 定义 IR 的指令集和操作语义
3. 实现从抽象语法树到 IR 的转换
4. 实现 IR 到目标代码的生成

**注意事项**:
- 保持 IR 的简洁性和正交性
- 确保 IR 能够表示所有源语言构造
- 考虑后续优化实现的便利性

---

### 实践 6：建立版本控制与代码审查机制

**说明**:
使用 AI 生成的大量代码需要有效的管理。通过版本控制系统跟踪每次变更，并建立代码审查流程，特别是对 AI 生成的关键代码进行人工审查，可以确保代码质量和可维护性。

**实施步骤**:
1. 初始化 Git 仓库并建立分支策略
2. 为每个智能体分配专门的工作分支
3. 实施合并前的代码审查流程
4. 保留所有重要决策的文档记录

**注意事项**:
- 提交信息应清晰描述变更内容
- 审查重点放在关键路径和复杂逻辑上
- 定期同步各智能体的工作成果

---

### 实践 7：设置明确的错误处理与恢复机制

**说明**:
编译器需要优雅地处理源代码中的错误。设计全面的错误检测、报告和恢复机制，确保编译器在遇到错误时能够提供有用的诊断信息，并尽可能继续分析以发现更多错误。

**实施步骤**:
1. 定义错误类型和严重级别
2. 实现错误消息格式化系统
3. 设计错误恢复策略（如同步点）
4. 测试各种错误场景的处理

**注意事项**:
- 错误消息应指向具体的源代码位置
- 避免级联错误（一个错误导致多个虚假错误）
