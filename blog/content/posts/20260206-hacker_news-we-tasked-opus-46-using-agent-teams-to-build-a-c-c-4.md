```python
# 示例1：词法分析器 - 将源代码分解为Token
import re

class Lexer:
    """简单的C语言词法分析器，用于识别关键字、标识符、数字和运算符"""
    
    # 定义Token类型
    KEYWORDS = {'int', 'return', 'if', 'else', 'while', 'for'}
    
    def __init__(self, text):
        self.text = text
        self.pos = 0
        self.current_char = self.text[self.pos] if self.text else None
    
    def advance(self):
        """移动到下一个字符"""
        self.pos += 1
        self.current_char = self.text[self.pos] if self.pos < len(self.text) else None
    
    def skip_whitespace(self):
        """跳过空白字符"""
        while self.current_char is not None and self.current_char.isspace():
            self.advance()
    
    def get_next_token(self):
        """获取下一个Token"""
        while self.current_char is not None:
            if self.current_char.isspace():
                self.skip_whitespace()
                continue
            
            # 识别标识符和关键字
            if self.current_char.isalpha() or self.current_char == '_':
                start = self.pos
                while self.current_char is not None and (self.current_char.isalnum() or self.current_char == '_'):
                    self.advance()
                value = self.text[start:self.pos]
                return ('KEYWORD', value) if value in self.KEYWORDS else ('ID', value)
            
            # 识别数字
            if self.current_char.isdigit():
                start = self.pos
                while self.current_char is not None and self.current_char.isdigit():
                    self.advance()
                return ('NUMBER', self.text[start:self.pos])
            
            # 识别运算符和分隔符
            if self.current_char in '+-*/=();{}':
                char = self.current_char
                self.advance()
                return ('OP', char) if char in '+-*/=' else ('DELIM', char)
            
            raise Exception(f'非法字符: {self.current_char}')
        
        return ('EOF', None)

# 测试
lexer = Lexer("int main() { return 0; }")
tokens = []
while True:
    token = lexer.get_next_token()
    tokens.append(token)
    if token[0] == 'EOF':
        break
print(tokens)  # 输出: [('KEYWORD', 'int'), ('ID', 'main'), ('DELIM', '('), ...]
```

```python
# 示例2：抽象语法树(AST) - 表示代码结构
class ASTNode:
    """抽象语法树节点基类"""
    pass

class BinOp(ASTNode):
    """二元运算节点，如 a + b"""
    def __init__(self, left, op, right):
        self.left = left
        self.op = op
        self.right = right
    
    def __repr__(self):
        return f'({self.left} {self.op} {self.right})'

class Num(ASTNode):
    """数字常量节点"""
    def __init__(self, value):
        self.value = value
    
    def __repr__(self):
        return str(self.value)

class Var(ASTNode):
    """变量节点"""
    def __init__(self, name):
        self.name = name
    
    def __repr__(self):
        return self.name

class Function(ASTNode):
    """函数定义节点"""
    def __init__(self, name, params, body):
        self.name = name
        self.params = params
        self.body = body
    
    def __repr__(self):
        return f'func {self.name}() {{ {self.body} }}'

# 构建一个简单的AST: int add(int a, int b) { return a + b; }
ast = Function(
    name='add',
    params=['a', 'b'],
    body=BinOp(Var('a'), '+', Var('b'))
)

print(ast)  # 输出: func add() { (a + b) }
```

class StackVM:
    """简单的栈式虚拟机，用于执行编译后的字节码"""
    
    def __init__(self):
        self.stack = []
        self.instructions = []
    
    def emit(self, op, *args):
        """生成指令"""
        self.instructions.append((op, args))
    
    def run(self):
        """执行字节码"""
        for op, args in self.instructions:
            if op == 'PUSH':
                self.stack.append(args[0])
            elif op == 'ADD':
                b = self.stack.pop()
                a = self.stack.pop()
                self.stack.append(a + b)
            elif op == 'SUB':
                b = self.stack.pop()
                a = self.stack.pop()
                self.stack.append(a - b)
            elif op == 'MUL':
                b = self.stack.pop()
                a = self.stack.pop()
                self.stack.append(a * b)
            elif op == 'PRINT':
                print(self.stack[-1])

## 最佳实践指南

### 实践 1：采用多智能体协作架构

**说明**: 在构建复杂系统（如 C 编译器）时，利用 AI 智能体团队代替单一模型。通过将编译器的不同组件（如词法分析、语法分析、代码生成、优化器）分配给专门的子智能体，可以显著提高系统的模块化程度和代码质量。

**实施步骤**:
1. 定义编译器的各个阶段和模块。
2. 为每个模块创建一个具有特定角色和职责的智能体。
3. 建立智能体之间的通信协议和数据传递标准。
4. 设置一个“主控”智能体负责协调各子智能体的工作进度和依赖关系。

**注意事项**: 确保各智能体之间的接口定义清晰，避免数据格式不兼容导致的集成问题。

---

### 实践 2：建立严格的测试驱动开发流程

**说明**: 编译器对正确性要求极高。在编写任何生成代码的逻辑之前，必须先编写测试用例。利用 AI 智能体生成针对特定 C 语言特性的测试用例，并验证编译器输出的汇编代码或机器码是否符合预期。

**实施步骤**:
1. 收集或生成标准 C 语言测试集（如 GCC 测试套件的一部分）。
2. 为每个编译器模块（如解析器、代码生成器）配置单元测试。
3. 要求智能体在提交代码前必须通过相关测试用例。
4. 引入模糊测试工具，随机生成 C 代码以发现边缘情况和崩溃漏洞。

**注意事项**: 测试覆盖率应作为核心指标，不仅要覆盖正常语法，还要覆盖错误处理和边缘情况。

---

### 实践 3：实施分阶段验证与中间表示检查

**说明**: 不要仅依赖最终生成的二进制文件进行验证。应在编译的每个阶段（如生成 AST 后、生成中间表示 IR 后）插入检查点，确保中间状态的正确性。

**实施步骤**:
1. 定义标准化的中间表示（IR）格式。
2. 编写验证器脚本，检查 AST 和 IR 的结构完整性。
3. 在智能体工作流中集成这些验证器，一旦某个阶段输出异常，立即回滚并修复。
4. 可视化中间结果，便于人工审查。

**注意事项**: 中间表示的设计应尽量简化，以便于智能体理解和操作，同时保留足够的高级语言信息。

---

### 实践 4：利用现有规范作为上下文基准

**说明**: C 语言标准（如 ISO C99 或 C11）是构建编译器的唯一真理来源。在提示词中明确引用标准文档，要求智能体严格遵循标准规范，而不是依赖训练数据中的模糊记忆。

**实施步骤**:
1. 准备 C 语言标准文档的文本片段。
2. 在任务描述中，将具体的标准条款作为上下文输入给智能体。
3. 针对复杂的语法规则（如指针算术或预处理指令），提供标准中的示例代码。

**注意事项**: 标准文档篇幅巨大，应根据当前任务动态检索相关的章节，避免超出上下文窗口限制。

---

### 实践 5：模块化设计与增量式开发

**说明**: 不要试图一次性生成整个编译器。应采用增量式开发策略，从简单的子集（如支持变量声明和基本运算）开始，逐步增加对控制流、函数、指针和复杂数据类型的支持。
