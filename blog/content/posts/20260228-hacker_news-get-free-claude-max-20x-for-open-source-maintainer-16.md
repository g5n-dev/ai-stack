# 示例3：监控GitHub项目星标变化并提醒申请
import requests
from datetime import datetime, timedelta

def check_and_notify(repo_owner, repo_name, last_check_time):
    """
    检查项目星标数是否达到申请门槛
    :param repo_owner: 仓库所有者
    :param repo_name: 仓库名称
    :param last_check_time: 上次检查时间
    :return: 是否应该申请
    """
    # 获取项目信息
    url = f"https://api.github.com/repos/{repo_owner}/{repo_name}"
    response = requests.get(url)
    data = response.json()
    
    stars = data['stargazers_count']
    updated_at = datetime.strptime(data['updated_at'], "%Y-%m-%dT%H:%M:%SZ")
    
    # 检查条件：星标数>1000 且 最近有更新
    should_apply = stars > 1000 and (datetime.utcnow() - updated_at) < timedelta(days=30)
    
    if should_apply:
        print(f"项目 {repo_owner}/{repo_name} 现有 {stars} 星标，建议申请Claude免费额度！")
    else:
        print(f"项目 {repo_owner}/{repo_name} 现有 {stars} 星标，暂不符合条件")
    
    return should_apply

# 使用示例
check_and_notify("python", "cpython", datetime.utcnow() - timedelta(days=7))
