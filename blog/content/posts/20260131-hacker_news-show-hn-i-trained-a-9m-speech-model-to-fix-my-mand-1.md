```python
# 示例1：提取并可视化中文拼音声调
import matplotlib.pyplot as plt
import numpy as np

def visualize_tones(pinyin_with_tone):
    """
    可视化拼音的声调模式（1-4声）
    参数:
        pinyin_with_tone: 带声调的拼音字符串，如 "mā"
    """
    # 定义声调对应的音高曲线
    tone_curves = {
        '1': [0.8, 0.8, 0.8, 0.8],  # 阴平（高平）
        '2': [0.4, 0.6, 0.8, 0.9],  # 阳平（升调）
        '3': [0.5, 0.3, 0.2, 0.4],  # 上声（降升）
        '4': [0.9, 0.7, 0.4, 0.2]   # 去声（降调）
    }
    
    # 提取声调数字（假设输入格式为 "ma1"）
    tone_num = pinyin_with_tone[-1]
    curve = tone_curves.get(tone_num, [0.5]*4)
    
    # 绘制声调曲线
    plt.figure(figsize=(6, 3))
    plt.plot(curve, 'o-', linewidth=2)
    plt.title(f"声调模式: {pinyin_with_tone}")
    plt.ylim(0, 1)
    plt.ylabel("音高")
    plt.grid(True)
    plt.show()

# 测试
visualize_tones("ma3")  # 上声示例
```

```python
# 示例2：基于规则修正声调错误
def correct_tone(text, corrections):
    """
    根据预定义规则修正文本中的声调错误
    参数:
        text: 输入的带声调拼音文本
        corrections: 修正规则字典，如 {"ma3": "ma4"}
    """
    words = text.split()
    corrected = []
    
    for word in words:
        # 检查是否需要修正
        if word in corrections:
            corrected.append(corrections[word])
        else:
            corrected.append(word)
    
    return " ".join(corrected)

# 测试
wrong_text = "wo3 xiang3 chi1 fan4"
corrections = {"fan4": "fan1"}  # 修正"饭"的声调
print(correct_tone(wrong_text, corrections))  # 输出: wo3 xiang3 chi1 fan1
```

```python
# 示例3：声调统计与分析
from collections import Counter

def analyze_tone_distribution(text):
    """
    分析文本中各声调的出现频率
    参数:
        text: 带声调的拼音文本
    """
    # 提取所有声调数字
    tones = [word[-1] for word in text.split() if word[-1].isdigit()]
    
    # 统计频率
    tone_counts = Counter(tones)
    total = sum(tone_counts.values())
    
    # 打印统计结果
    print("声调分布统计:")
    for tone in ['1', '2', '3', '4']:
        count = tone_counts.get(tone, 0)
        percent = count/total*100 if total > 0 else 0
        print(f"{tone}声: {count}次 ({percent:.1f}%)")

# 测试
sample_text = "ni3 hao3 ma1 wo3 hen3 hao3 xie4 xie4"
analyze_tone_distribution(sample_text)
```

## 最佳实践指南

### 实践 1：构建针对性的声调数据集

**说明**: 普通话的四个声调（外加轻声）是语义区别的关键特征。通用的语音识别数据集往往缺乏对声调细节的精细标注，或者混合了不同口音。训练一个专门用于纠正声调的模型，首要任务是准备一个高质量、声调标注准确的特定数据集，而非仅仅依赖大规模的通用网络抓取数据。

**实施步骤**:
1. 收集标准普通话的音频源，优先选择新闻广播、有声书或专业配音素材。
2. 使用强制对齐工具或人工校验，确保音素级别的声调标签与音频波形精确对应。
3. 对数据进行清洗，去除背景噪音和带有浓重方言口音的样本，确保声调特征的纯净度。

**注意事项**: 必须验证数据集中说话人的声调覆盖度，避免某些声调（如第三声）的数据样本过少导致模型偏见。

---

### 实践 2：采用轻量化模型架构

**说明**: 对于声调纠正这类特定任务，参数量并非越大越好。9M（9百万）参数的模型证明了在特定垂直领域，小模型配合良好的训练策略可以达到优于大模型的泛化能力和推理速度。轻量化模型更适合部署在移动端或边缘设备，实现实时反馈。

**实施步骤**:
1. 选择适合小样本学习的架构，如精简版的 Conformer 或 Efficient Transformer 变体。
2. 减少模型的深度和宽度，但保留足够的注意力头以捕捉音频中的长距离依赖关系。
3. 使用知识蒸馏技术，如果存在更大的教师模型，将其知识迁移到 9M 的学生模型中。

**注意事项**: 在缩减模型大小时，要平衡模型容量与任务复杂度，确保模型有足够的容量去捕捉声调的细微韵律变化。

---

### 实践 3：利用自监督学习进行预训练

**说明**: 监督学习的数据往往有限。利用自监督学习（如 wav2vec 2.0 或 HuBERT）在大量无标注普通话语音上进行预训练，可以让模型学习到声调的内在声学特征和语言结构。这能显著提升模型在少量标注数据上的收敛速度和最终准确率。

**实施步骤**:
1. 收集大规模的无标注普通话音频数据。
2. 使用掩码语音预测任务进行预训练，让模型学会从上下文中重建被掩盖的音频特征。
3. 在预训练权重基础上，使用带声调标签的特定数据集进行微调。

**注意事项**: 预训练数据的多样性至关重要，应包含不同性别、年龄和轻微的语速变化，以防止过拟合。
