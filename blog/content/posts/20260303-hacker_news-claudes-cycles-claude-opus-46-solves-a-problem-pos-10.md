# 示例1：生成所有排列组合
def generate_permutations(elements):
    """
    生成给定元素的所有可能排列组合
    使用递归算法实现（Knuth算法T的简化版）
    
    参数:
        elements: 可迭代对象（如列表、字符串）
    
    返回:
        包含所有排列的列表
    """
    if len(elements) <= 1:
        return [elements]
    
    result = []
    for i in range(len(elements)):
        # 固定当前元素，递归处理剩余元素
        current = elements[i]
        remaining = elements[:i] + elements[i+1:]
        for p in generate_permutations(remaining):
            result.append([current] + p)
    
    return result

# 测试用例
if __name__ == "__main__":
    print("ABC的排列组合:")
    for perm in generate_permutations(['A', 'B', 'C']):
        print(perm)
```

def find_cyclic_permutations(sequence):
    """
    找出给定序列的所有循环排列（旋转排列）
    例如：[1,2,3]的循环排列包括[1,2,3], [2,3,1], [3,1,2]
    
    参数:
        sequence: 可迭代对象
    
    返回:
        包含所有循环排列的列表
    """
    n = len(sequence)
    result = []
    
    for i in range(n):
        # 通过切片实现循环移位
        rotated = sequence[i:] + sequence[:i]
        result.append(rotated)
    
    return result

# 测试用例
if __name__ == "__main__":
    print("\n[1,2,3,4]的循环排列:")
    for perm in find_cyclic_permutations([1, 2, 3, 4]):
        print(perm)
```

# 示例3：检测排列中的循环结构
def detect_cycles(permutation):
    """
    检测一个排列中的循环结构
    例如：排列[2,3,1,4]可以分解为循环(1 2 3)和(4)
    
    参数:
        permutation: 表示排列的列表（从1开始编号）
    
    返回:
        包含所有循环的列表，每个循环表示为元组
    """
    n = len(permutation)
    visited = [False] * (n + 1)  # 记录已访问的元素
    cycles = []
    
    for i in range(1, n + 1):
        if not visited[i]:
            cycle = []
            current = i
            while not visited[current]:
                visited[current] = True
                cycle.append(current)
                current = permutation[current - 1]  # 转换为0-based索引
            if len(cycle) > 1:
                cycles.append(tuple(cycle))
    
    return cycles

# 测试用例
if __name__ == "__main__":
    print("\n排列[2,3,1,4]的循环结构:")
    cycles = detect_cycles([2, 3, 1, 4])
    for cycle in cycles:
        print(f"循环: {cycle}")
