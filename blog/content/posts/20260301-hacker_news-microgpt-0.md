def simple_text_generation():
    """
    使用 MicroGPT 生成简单的文本内容
    解决问题：快速生成类似 GPT 的文本补全
    """
    from microgpt import MicroGPT
    
    # 初始化模型（使用默认配置）
    model = MicroGPT()
    
    # 输入提示词
    prompt = "今天天气很好，适合去"
    
    # 生成文本（设置生成长度和温度参数）
    output = model.generate(
        prompt,
        max_length=20,  # 最多生成20个token
        temperature=0.7  # 控制随机性（0-1，越高越随机）
    )
    
    print(f"输入: {prompt}")
    print(f"生成: {output}")

1: MicroGPT 是什么？它与 ChatGPT 等大型模型有何区别？
