def risk_assessment(transaction_data):

def optimize_portfolio(current_holdings, market_data):
    """
    基于市场数据提供投资组合优化建议
    参数: current_holdings - 当前持仓组合
           market_data - 市场数据（价格、波动率等）
    返回: 优化建议和调整方案
    """
    # 这里模拟Claude的分析能力
    # 实际应用中可以结合Claude对市场趋势的判断
    
    # 计算当前组合特征
    total_value = sum(h['amount'] * market_data[h['symbol']]['price'] 
                     for h in current_holdings)
    
    # 简单的再平衡逻辑（实际应用会更复杂）
    recommendations = []
    for holding in current_holdings:
        weight = (holding['amount'] * market_data[holding['symbol']]['price']) / total_value
        if weight > 0.4:  # 单只持仓超过40%建议分散
            recommendations.append({
                'symbol': holding['symbol'],
                'action': 'reduce',
                'reason': f'当前持仓{weight:.1%}过高，建议降低至30%以下'
            })
    
    return {
        '总市值': total_value,
        '调整建议': recommendations,
        '风险提示': '市场波动较大，建议分批调整'
    }

# 测试示例

 1：全球性投资银行自动化财报分析

**解决方案**:

**效果**:
