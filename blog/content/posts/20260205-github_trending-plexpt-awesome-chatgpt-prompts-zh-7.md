```python
def process_text_files(input_folder, output_folder):
    """
    批量处理文本文件：将输入文件夹中的所有txt文件转换为大写并保存到输出文件夹
    :param input_folder: 输入文件夹路径
    :param output_folder: 输出文件夹路径
    """
    import os
    
    # 确保输出文件夹存在
    os.makedirs(output_folder, exist_ok=True)
    
    # 遍历输入文件夹中的所有文件
    for filename in os.listdir(input_folder):
        if filename.endswith('.txt'):
            input_path = os.path.join(input_folder, filename)
            output_path = os.path.join(output_folder, f"processed_{filename}")
            
            # 读取文件内容并转换为大写
            with open(input_path, 'r', encoding='utf-8') as f:
                content = f.read().upper()
            
            # 写入处理后的内容
            with open(output_path, 'w', encoding='utf-8') as f:
                f.write(content)
            
            print(f"已处理文件: {filename}")

# 使用示例
process_text_files("input_texts", "output_texts")
```

```python
# 示例2：生成随机密码
def generate_password(length=12, include_symbols=True):
    """
    生成指定长度的随机密码
    :param length: 密码长度，默认12
    :param include_symbols: 是否包含特殊符号，默认True
    :return: 生成的密码字符串
    """
    import random
    import string
    
    # 定义字符集
    letters = string.ascii_letters  # 大小写字母
    digits = string.digits          # 数字
    symbols = string.punctuation    # 特殊符号
    
    # 根据参数选择字符集
    chars = letters + digits
    if include_symbols:
        chars += symbols
    
    # 确保密码包含各种字符类型
    password = [
        random.choice(letters),
        random.choice(digits),
        random.choice(symbols) if include_symbols else random.choice(letters)
    ]
    
    # 填充剩余长度
    password += [random.choice(chars) for _ in range(length - len(password))]
    
    # 打乱顺序并转换为字符串
    random.shuffle(password)
    return ''.join(password)

# 使用示例
print(generate_password(16))  # 生成16位包含符号的密码
print(generate_password(8, False))  # 生成8位不含符号的密码
```

```python
# 示例3：简单的Web爬虫
def simple_web_scraper(url, output_file):
    """
    简单的网页爬虫，提取网页中的所有链接并保存到文件
    :param url: 要爬取的网页URL
    :param output_file: 输出文件路径
    """
    import requests
    from bs4 import BeautifulSoup
    
    try:
        # 发送HTTP请求
        response = requests.get(url)
        response.raise_for_status()  # 检查请求是否成功
        
        # 解析HTML内容
        soup = BeautifulSoup(response.text, 'html.parser')
        
        # 提取所有链接
        links = []
        for link in soup.find_all('a', href=True):
            href = link['href']
            if href.startswith('http'):  # 只保存绝对路径
                links.append(href)
        
        # 保存到文件
        with open(output_file, 'w', encoding='utf-8') as f:
            for link in links:
                f.write(f"{link}\n")
        
        print(f"成功提取 {len(links)} 个链接，已保存到 {output_file}")
        
    except Exception as e:
        print(f"爬取失败: {str(e)}")

# 使用示例
simple_web_scraper("https://example.com", "links.txt")
```

## 最佳实践

## 提示词工程最佳实践

### 1. 明确角色设定
**核心原则**：赋予模型具体的专业身份，以确立回答的基调和深度。
*   **操作方法**：使用“你是一位[具体角色]...”的句式。
*   **示例**：不要只说“你是专家”，而应说“你是一位拥有10年经验的Python后端工程师，擅长高并发架构设计”。

### 2. 结构化指令
**核心原则**：将复杂任务拆解为清晰的步骤，并明确输出格式。
*   **操作方法**：使用Markdown列表，明确指定输出形式（如表格、JSON、代码块）。
*   **关键点**：确保步骤逻辑连贯，避免指令冲突。

### 3. 补充背景上下文
**核心原则**：提供充分的背景信息（目标、受众、限制），以减少歧义。
*   **操作方法**：说明“为什么要做”以及“给谁看”。
*   **关键点**：保持信息精简，剔除无关的噪音数据。

### 4. 设置约束条件
**核心原则**：通过明确的规则和限制来控制输出的范围和质量。
*   **操作方法**：列出“必须做”和“禁止做”的事项，设定量化指标（如字数、段落数）。
*   **示例**：“回复不超过200字，且不使用任何被动语态”。

### 5. 迭代优化
**核心原则**：将提示词工程视为一个持续优化的过程，而非一次性任务。
*   **操作方法**：基于输出结果进行评估，指出偏差，调整指令后重新生成。
*   **关键点**：保持核心目标的一致性，逐步微调细节。

### 6. 提供少样本示例
**核心原则**：通过“输入-输出”示例，直观地展示期望的格式和风格。
*   **操作方法**：在提示词中包含1-3个典型示例。
*   **关键点**：示例必须与当前任务高度相关，避免误导模型。

### 7. 验证与测试
**核心原则**：在不同场景下系统性测试提示词的鲁棒性。
*   **操作方法**：设计包含边界情况的测试用例，记录并分析结果差异。
*   **关键点**：确保提示词在处理异常输入时仍能保持稳定。

### 实践建议：高效利用 ChatGPT 指令库

1.  **精准角色设定**
    在提问时明确赋予 AI 具体的专家身份（如“资深 Python 工程师”或“拥有 10 年经验的文案策划”）。这种“人设注入”能锁定专业语境，有效减少幻觉，显著提升输出的专业度与逻辑性。

2.  **应用少样本提示**
    不要仅描述需求，直接在提示词中提供 1-2 个具体的“输入-输出”示例。通过展示期望的问答对，让模型迅速模仿特定的语气、结构和风格，比纯文字描述更准确。

3.  **严格输出约束**

4.  **构建思维链**
    面对复杂逻辑或编程任务，加入“请一步步思考”或“分步骤解决问题”的指令。引导模型展示推理过程，能大幅降低逻辑跳跃和错误的发生率。

5.  **迭代式对话**
    将 ChatGPT 视作实习生：先要求生成大纲或草稿，再针对不足进行定向修正（如“替换第三点为更具体的数据”）。通过多轮反馈，内容质量将呈指数级提升。
