# 示例1：计算量子纠缠态的保真度
def calculate_entanglement_fidelity(state1, state2):
    """
    计算两个量子纠缠态之间的保真度（相似度）
    
    参数:
        state1 (list): 第一个量子态的复数振幅数组
        state2 (list): 第二个量子态的复数振幅数组
        
    返回:
        float: 保真度值（0到1之间，1表示完全相同）
    """
    import numpy as np
    
    # 转换为numpy数组方便计算
    state1 = np.array(state1)
    state2 = np.array(state2)
    
    # 计算内积
    inner_product = np.vdot(state1, state2)
    
    # 保真度是内积模的平方
    fidelity = np.abs(inner_product)**2
    
    return fidelity

# 测试用例
if __name__ == "__main__":
    # 定义两个量子态
    psi = [1/np.sqrt(2), 1/np.sqrt(2)]  # 叠加态
    phi = [1, 0]  # 基态
    
    print(f"保真度: {calculate_entanglement_fidelity(psi, phi):.4f}")
```

# 示例2：模拟黑洞霍金辐射的温度计算
def hawking_temperature(mass_solar_units):
    """
    计算黑洞的霍金辐射温度
    
    参数:
        mass_solar_units (float): 黑洞质量（以太阳质量为单位）
        
    返回:
        float: 霍金温度（开尔文）
    """
    import scipy.constants as const
    
    # 物理常数
    G = const.G  # 万有引力常数
    h = const.h  # 普朗克常数
    c = const.c  # 光速
    k = const.k  # 玻尔兹曼常数
    
    # 太阳质量(kg)
    solar_mass = 1.989e30
    
    # 黑洞质量(kg)
    mass = mass_solar_units * solar_mass
    
    # 霍金温度公式
    T = (h * c**3) / (8 * np.pi * G * mass * k)
    
    return T

# 测试用例
if __name__ == "__main__":
    import numpy as np
    
    # 计算不同质量黑洞的温度
    masses = [1, 10, 100]  # 太阳质量
    for m in masses:
        print(f"{m}倍太阳质量黑洞的温度: {hawking_temperature(m):.2e} K")
```

# 示例3：宇宙微波背景辐射(CMB)功率谱分析
def cmb_power_spectrum(ell, A=1e-10, l_peak=200):
    """
    计算宇宙微波背景辐射的角功率谱
    
    参数:
        ell (array): 多极矩数组
        A (float): 幅度参数
        l_peak (float): 第一声学峰位置
        
    返回:
        array: 功率谱值
    """
    import numpy as np
    
    # 简化的功率谱模型
    Cl = A * (ell / l_peak)**2 / (1 + (ell / l_peak)**2)**2
    
    return Cl

# 测试用例
if __name__ == "__main__":
    import matplotlib.pyplot as plt
    
    # 生成多极矩范围
    ell = np.linspace(2, 1000, 1000)
    
    # 计算功率谱
    Cl = cmb_power_spectrum(ell)
    
    # 绘制功率谱
    plt.figure(figsize=(10, 6))
    plt.loglog(ell, ell*(ell+1)*Cl/(2*np.pi))
    plt.xlabel('多极矩 $\ell$')
    plt.ylabel('$\ell(\ell+1)C_\ell/2\pi$')
    plt.title('宇宙微波背景辐射功率谱')
    plt.grid(True)
    plt.show()
