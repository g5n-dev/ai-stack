# 示例1：词法分析器
# 功能：将C语言源代码字符串分解为有意义的标记流
import re

class Token:
    """表示一个词法标记"""
    def __init__(self, type_, value):
        self.type = type_    # 标记类型
        self.value = value   # 标记值
    
    def __repr__(self):
        return f"Token({self.type}, {repr(self.value)})"

def tokenize(source_code):
    """
    词法分析函数
    :param source_code: C语言源代码字符串
    :return: 标记列表
    """
    # 定义C语言的关键字
    keywords = {'int', 'float', 'return', 'if', 'else', 'while', 'for', 'void'}
    
    # 定义正则表达式规则
    token_specification = [
        ('NUMBER',   r'\d+(\.\d*)?'),  # 整数或小数
        ('ASSIGN',   r'='),           # 赋值运算符
        ('END',      r';'),           # 语句结束符
        ('OP',       r'[+\-*/]'),     # 算术运算符
        ('ID',       r'[A-Za-z_]\w*'),# 标识符
        ('SKIP',     r'[ \t]+'),      # 跳过空格和制表符
        ('NEWLINE',  r'\n'),          # 换行符
        ('MISMATCH', r'.'),           # 其他不匹配的字符
    ]
    
    # 编译正则表达式
    tok_regex = '|'.join('(?P<%s>%s)' % pair for pair in token_specification)
    get_token = re.compile(tok_regex).match
    
    tokens = []
    pos = 0
    mo = get_token(source_code)
    
    while mo is not None:
        kind = mo.lastgroup
        value = mo.group()
        
        if kind == 'NUMBER':
            token = Token(kind, float(value) if '.' in value else int(value))
        elif kind == 'ID':
            # 检查是否为关键字
            token_type = 'KEYWORD' if value in keywords else 'ID'
            token = Token(token_type, value)
        elif kind not in ('SKIP', 'NEWLINE', 'MISMATCH'):
            token = Token(kind, value)
        else:
            token = None
        
        if token:
            tokens.append(token)
        
        pos = mo.end()
        mo = get_token(source_code, pos)
    
    return tokens

# 测试代码
code = "int x = 42; float y = 3.14;"
print(tokenize(code))

        self.right = right  # 右操作数

    """数字字面量节点"""
    def __init__(self, token):
        self.token = token
        self.value = token.value

class Assign(ASTNode):
    """赋值语句节点"""
    def __init__(self, left, op, right):
        self.left = left    # 左值(变量)
        self.op = op        # 赋值运算符
        self.right = right  # 右值(表达式)

class Var(ASTNode):
    """变量节点"""
    def __init__(self, token):
        self.token = token
        self.value = token.value

class Parser:

        self.pos = 0

## 最佳实践指南

### 实践 1：采用多智能体协作架构

**说明**: 利用 AI 模型（如 Opus 4.6）构建复杂系统时，不应将其视为单一编码器，而应组建专门的智能体团队。通过分配不同的角色（如架构师、核心逻辑开发者、测试工程师、文档编写者），可以模拟真实的软件开发团队，从而解决单一上下文窗口无法容纳大型项目全貌的问题。

**实施步骤**:
1. 定义角色分工：设立“产品经理”负责需求拆解，“架构师”负责设计模块接口，“开发人员”负责具体函数实现，“QA 工程师”负责编写测试用例。
2. 建立通信协议：明确各智能体之间的输入输出格式，确保代码和文档在不同智能体间无缝传递。
3. 设置工作流：确定是采用并行开发（不同模块同时进行）还是串行开发（先完成词法分析，再进行语法分析）。

**注意事项**: 避免智能体之间形成“回音室”效应，即错误信息在智能体间相互传递并被放大。需要引入独立的审查机制。

---

### 实践 2：实施模块化与增量式开发

**说明**: C 编译器是一个高度复杂的系统，包含词法分析、语法分析、语义分析、中间代码生成和优化等阶段。试图一次性生成整个编译器是不现实的。最佳实践是遵循编译原理的教科书式步骤，将其拆解为独立的模块，并按顺序逐步构建。

**实施步骤**:
1. 第一阶段：构建词法分析器，将源代码转换为 Token 流。
2. 第二阶段：构建语法分析器，基于递归下降或 LR 算法构建抽象语法树（AST）。
3. 第三阶段：实现语义分析，确保类型安全和变量声明合规。
4. 第四阶段：实现代码生成，将 AST 转换为目标汇编代码。

**注意事项**: 每个模块完成后，必须通过独立的测试集验证，确保其输出符合下一级模块的输入要求，防止“垃圾进，垃圾出”。

---

### 实践 3：建立“测试驱动”的验证循环

**说明**: 在 AI 生成代码的过程中，产生逻辑错误或幻觉是常见的。对于编译器这类对正确性要求极高的软件，必须依赖自动化测试套件来验证每一阶段生成的代码。不能仅依赖 AI 的自我检查，必须通过编译和运行真实的 C 代码来验证。

**实施步骤**:
1. 准备基准测试集：收集一系列简单的 C 语言代码片段（如 Hello World, 斐波那契数列计算等）作为“黄金标准”。
2. 编写断言测试：针对编译器的各个阶段编写单元测试，例如检查 Token 序列是否正确，AST 结构是否符合预期。
3. 持续集成：每当 AI 生成新的模块代码时，立即运行测试套件，如果测试失败，将错误信息反馈给 AI 进行修复。

**注意事项**: 测试用例应从最基础的功能开始，随着功能的增加逐渐扩大测试覆盖范围，避免在早期阶段使用过于复杂的代码导致调试困难。

---

### 实践 4：利用形式化规范与标准参考

**说明**: AI 模型虽然训练了大量代码，但在处理边缘情况或特定的 C 语言标准（如 C89, C99, C11）时可能会产生混淆。提供明确的标准文档（如 ISO C 标准）或形式化语法规则（如 BNF 范式）作为上下文，可以显著提高代码的准确性。

**实施步骤**:
1. 在 Prompt 中明确指定遵循的 C 语言标准版本（例如：“请严格遵循 C11 标准”）。
2. 提供关键数据结构的定义或接口规范，强制 AI 按照预定义的接口进行实现。
3. 将复杂的算法逻辑（如寄存器分配算法）以伪代码形式提供给 AI，让其翻译成具体代码，而非让其从头构思算法。

**注意事项**: 确保提供给 AI 的参考文档是准确的，且与当前开发阶段的目标一致，以免造成误导。

---

### 实践 5：构建人机协同的调试机制

**说明**: 尽管 AI 智能体团队可以自动完成大部分工作，但在遇到深层逻辑错误或性能瓶颈时，人类的介入至关重要。开发者应扮演“技术主管”的角色，审查关键代码，并指导 AI 进行修复。

**实施步骤**:
