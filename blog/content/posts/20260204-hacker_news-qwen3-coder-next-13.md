# 示例1：自动化文件整理
import os
import shutil
from pathlib import Path

def organize_files(folder_path):
    """
    将指定文件夹中的文件按扩展名分类到子文件夹中
    实际应用：整理下载文件夹，自动分类文档、图片、视频等
    """
    folder = Path(folder_path)
    file_types = {
        '文档': ['.pdf', '.doc', '.docx', '.txt'],
        '图片': ['.jpg', '.jpeg', '.png', '.gif'],
        '视频': ['.mp4', '.avi', '.mov'],
        '音频': ['.mp3', '.wav']
    }
    
    for file in folder.iterdir():
        if file.is_file():
            ext = file.suffix.lower()
            for category, extensions in file_types.items():
                if ext in extensions:
                    target_dir = folder / category
                    target_dir.mkdir(exist_ok=True)
                    shutil.move(str(file), str(target_dir / file.name))
                    print(f"已移动 {file.name} 到 {category} 文件夹")

# 使用示例
# organize_files("C:/Users/YourName/Downloads")

import requests
from bs4 import BeautifulSoup
import pandas as pd

def scrape_news_titles(url):
    """
    从指定URL抓取新闻标题并保存为CSV文件
    实际应用：自动收集行业新闻，生成简报
    """
    headers = {'User-Agent': 'Mozilla/5.0'}
    response = requests.get(url, headers=headers)
    soup = BeautifulSoup(response.text, 'html.parser')
    
    # 提取新闻标题和链接（这里以Hacker News为例）
    news_data = []
    for item in soup.select('.storylink'):
        title = item.text.strip()
        link = item['href']
        news_data.append({'标题': title, '链接': link})
    
    # 保存为CSV文件
    df = pd.DataFrame(news_data)
    df.to_csv('news_report.csv', index=False, encoding='utf-8-sig')
    print(f"已保存 {len(news_data)} 条新闻到 news_report.csv")

# 使用示例

# 示例3：简单文本分析工具
from collections import Counter
import re

def analyze_text(text):
    """
    对输入文本进行基本分析：词频统计、句子数量、平均句长
    实际应用：快速评估文章可读性，或用于SEO关键词分析
    """
    # 清理文本：去除标点和特殊字符
    clean_text = re.sub(r'[^\w\s]', '', text.lower())
    
    # 分词（英文按空格，中文需要分词库）
    words = clean_text.split()
    
    # 统计词频
    word_freq = Counter(words)
    
    # 统计句子数量（简单按句号分割）
    sentences = text.split('.')
    sentence_count = len([s for s in sentences if s.strip()])
    
    # 计算结果
    result = {
        '总词数': len(words),
        '句子数量': sentence_count,
        '平均句长': len(words) / max(sentence_count, 1),
        '高频词': word_freq.most_common(5)
    }
    
    return result

# 使用示例
sample_text = """
Python is a versatile programming language. 
It is used for web development, data analysis, and automation.
Python has a simple syntax that is easy to learn.
"""
print(analyze_text(sample_text))

## 最佳实践指南

### 实践 1：构建高上下文感知的编程提示词

**说明**: Qwen3-Coder-Next 在处理代码任务时，对上下文信息的依赖度较高。简单的指令往往只能得到通用的代码，而详细的上下文（如项目结构、依赖库版本、特定变量定义）能显著提升代码的可用性和准确性。

**实施步骤**:
1. 在提问前，使用 Markdown 代码块明确提供相关的背景代码，不要仅用文字描述。
2. 明确指出代码的运行环境（如 Python 3.10, Node.js 18 等）和使用的框架。
3. 如果是调试任务，请务必包含完整的错误堆栈信息。

**注意事项**: 避免一次性粘贴过长的无关代码，仅保留与当前逻辑紧密相关的核心片段，以防止模型注意力分散。

---

### 实践 2：利用思维链技术进行复杂逻辑推理

**说明**: 对于算法设计或架构重构等复杂任务，直接要求生成代码可能导致逻辑漏洞。利用思维链让模型先进行逐步分析和规划，可以显著提高代码的健壮性。

**实施步骤**:
1. 在提示词中明确要求：“请先分析需求，列出步骤，再编写代码”。
2. 可以要求模型在生成代码前，先伪代码或流程图的形式展示逻辑。
3. 对生成的逻辑步骤进行确认后，再要求模型基于该步骤生成具体实现。

**注意事项**: 如果模型的推理过程出现偏差，应及时在对话中纠正其逻辑步骤，而不是直接要求重写代码。

---

### 实践 3：指定严格的代码风格与格式规范

**说明**: 为了确保生成的代码能够直接集成到现有的工程中，必须显式地约束代码风格。Qwen3-Coder-Next 对格式指令的遵循能力较强，可以利用这一点来统一代码标准。

**实施步骤**:
1. 在系统提示词或用户指令中明确编码规范，例如：“使用 Google Python Style Guide”或“使用 Airbnb JavaScript Style”。
2. 指定具体的缩进方式（如空格或 Tab）及宽度。
3. 明确要求文档字符串的标准，如：“请为所有函数添加 Google 风格的 Docstring”。

**注意事项**: 某些格式要求可能存在冲突（如同时要求单行代码长度限制和复杂逻辑），需根据实际情况权衡。

---

### 实践 4：实施交互式迭代与单元测试验证

**说明**: 一次性生成完美代码的概率较低。最佳实践是将 Qwen3-Coder-Next 作为一个结对编程伙伴，通过交互式的反馈循环来完善代码，并利用其生成测试用例的能力。

**实施步骤**:
1. 获取代码初稿后，不要直接复制，而是要求模型：“请为这段代码生成单元测试用例，覆盖边界条件”。
2. 运行测试用例，将失败的错误信息反馈给模型：“测试用例 3 失败了，错误信息是...，请修复代码”。
3. 要求模型对代码进行自我审查：“请检查这段代码是否存在潜在的性能瓶颈或安全漏洞”。

**注意事项**: 模型生成的测试用例可能并不完全覆盖所有逻辑死角，人工审查测试逻辑依然必要。

---

### 实践 5：针对特定语言生态进行微调提示

**说明**: Qwen3-Coder-Next 涵盖多种语言，但不同语言（如 Rust vs Python）的最佳开发范式差异巨大。针对特定语言调整提问方式，能获得更地道（Idiomatic）的代码。

**实施步骤**:
1. **针对 Rust/Go**: 强调“内存安全”、“并发处理”和“错误处理机制”，要求代码符合语言特有的所有权模型或接口惯例。
2. **针对 Python/JavaScript**: 强调“库的使用”和“简洁性”，要求利用现代语法特性（如 Python 的 Type Hints 或 JS 的 ES6+ 特性）。
3. **针对 SQL**: 明确数据库类型（如 PostgreSQL, MySQL），并要求优化查询性能。

**注意事项**: 避免要求模型使用过时的语法或库，如果不确定，可以询问：“目前解决该问题的最佳实践库是什么？”

---

### 实践 6：利用代码审查模式重构遗留代码

**说明**: 除了生成新代码，该模型非常擅长理解和重构旧代码。通过设定特定的审查目标，可以提升代码库的可维护性。

**实施步骤**:
1. 提供 Legacy Code，并设定角色：“你是一位资深的代码审查专家”。
2. 提出具体的重构目标，例如：“请在不改变功能逻辑的前提下，提高这段代码的可读性并降低圈复杂度”。
3. 要求解释修改理由：“请列出你做出的主要修改点及其原因”。

**注意事项**: 在重构关键业务逻辑时，务必要求模型保持输入输出的一致性，并严格对比测试结果。
