quantum_simulation()

def hawking_temperature(mass_solar_masses):
    """计算黑洞的霍金辐射温度 (开尔文)"""
    # 物理常数
    G = 6.674e-11  # 引力常数
    h_bar = 1.054e-34  # 约化普朗克常数
    c = 3e8  # 光速
    k_B = 1.38e-23  # 玻尔兹曼常数
    
    # 太阳质量转换为千克
    mass = mass_solar_masses * 1.989e30
    
    # 霍金温度公式 T = ħc³/(8πGMk_B)
    T = (h_bar * c**3) / (8 * np.pi * G * mass * k_B)
    return T

# 计算5倍太阳质量的黑洞温度
print(f"5倍太阳质量黑洞温度: {hawking_temperature(5):.2e} K")

def cmb_power_spectrum(ell, A=1e-10, l_peak=200):
    """简化版CMB角功率谱模型"""
    # 使用阻尼振荡模型拟合

from scipy.optimize import curve_fit

**注意事项**:
不要试图一次性验证整个理论，应采用分段验证策略，以便快速定位潜在的错误源头。

**注意事项**:

**注意事项**:
