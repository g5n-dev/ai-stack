import matplotlib.pyplot as plt
import numpy as np

def create_infographic():
    """创建一个包含多元素的专业信息图表"""
    plt.figure(figsize=(10, 6))
    
    # 数据准备
    categories = ['AI技术', '云计算', '大数据', '物联网', '区块链']
    values = [85, 72, 90, 68, 55]
    
    # 创建柱状图
    bars = plt.bar(categories, values, color=['#FF6B6B', '#4ECDC4', '#45B7D1', '#FFA07A', '#98D8C8'])
    
    # 添加数值标签
    for bar in bars:
        height = bar.get_height()
        plt.text(bar.get_x() + bar.get_width()/2., height,
                f'{height}%',
                ha='center', va='bottom', fontsize=12)
    
    # 设置标题和样式
    plt.title('2024年技术趋势调查', fontsize=16, pad=20)
    plt.ylabel('采用率 (%)', fontsize=12)
    plt.ylim(0, 100)
    plt.grid(axis='y', linestyle='--', alpha=0.7)
    
    # 保存为高清图片
    plt.savefig('infographic.png', dpi=300, bbox_inches='tight')
    plt.show()

# 调用函数生成图表
create_infographic()

from PIL import Image, ImageEnhance, ImageFilter

def enhance_photo(input_path, output_path):
    """对照片进行专业级增强处理"""
    # 打开原始图片
    img = Image.open(input_path)
    
    # 锐化处理
    enhancer = ImageEnhance.Sharpness(img)
    img = enhancer.enhance(1.5)
    
    # 色彩增强
    color = ImageEnhance.Color(img)
    img = color.enhance(1.2)
    
    # 对比度调整
    contrast = ImageEnhance.Contrast(img)
    img = contrast.enhance(1.1)
    
    # 应用轻微模糊滤镜
    img = img.filter(ImageFilter.SMOOTH_MORE)
    
    # 保存处理后的图片
    img.save(output_path, quality=95)
    print(f"处理后的图片已保存至: {output_path}")

# 使用示例
enhance_photo('input.jpg', 'enhanced_output.jpg')

# 示例3：生成式图像风格迁移
import torch
from torchvision import transforms
from models import TransformerNet  # 假设已预训练模型

def style_transfer(content_img_path, style_img_path, output_path):
    """将艺术风格迁移到内容图片上"""
    # 加载预训练模型
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model = TransformerNet().to(device)
    model.load_state_dict(torch.load('style_model.pth'))
    model.eval()
    
    # 图像预处理
    transform = transforms.Compose([
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.485, 0.456, 0.406],
                           std=[0.229, 0.224, 0.225])
    ])
    
    # 加载图像
    content = Image.open(content_img_path)
    content_tensor = transform(content).unsqueeze(0).to(device)
    
    # 风格迁移
    with torch.no_grad():
        output = model(content_tensor)
    
    # 保存结果
    output_img = transforms.ToPILImage()(output.squeeze(0).cpu())
    output_img.save(output_path)
    print(f"风格迁移结果已保存至: {output_path}")

# 使用示例
style_transfer('content.jpg', 'style.jpg', 'styled_output.jpg')

### 实践 1：构建结构化的信息图表提示词

**说明**: Qwen-Image-2.0 在生成专业信息图表方面表现出色，能够理解复杂的层级关系和数据可视化需求。为了获得最佳效果，必须使用结构化的提示词，明确指定图表类型、布局逻辑和包含的数据元素。

**实施步骤**:
1. 在提示词中明确定义图表类型（如：流程图、树状图、对比矩阵或时间轴）。
2. 使用清晰的分隔符或括号来组织文本层级，例如 `[标题: 季度增长]`, `[副标题: 财务数据]`。
3. 指定视觉流向，例如“从左到右”、“中心辐射”或“垂直堆叠”。

**注意事项**: 避免使用过于晦涩的缩写，确保所有标签和图例在生成的图片中清晰可读。

### 实践 2：利用摄影级写实指令增强质感

**说明**: 该模型在“exquisite photorealism”（精致的摄影写实）方面表现优异。通过在提示词中强调光影、材质和物理细节，可以生成具有极高欺骗性的真实感图像，适用于产品展示或场景渲染。

**实施步骤**:
1. 在描述主体后，附加具体的摄影术语，如“8k resolution”、“macro photography”（微距摄影）或“cinematic lighting”（电影级布光）。
2. 描述材质细节，例如“glass reflection”（玻璃反光）、“skin texture”（皮肤纹理）或“metallic sheen”（金属光泽）。
3. 指定环境光与主体互动的方式，如“volumetric lighting”（体积光）或“softbox lighting”（柔光箱效果）。

**注意事项**: 追求极致写实时，避免添加“cartoon”或“illustration”等冲突的风格词汇，以免导致模型风格混淆。

---

### 实践 3：精确控制文本渲染与排版

**说明**: 信息图表通常包含大量文字。虽然 Qwen-Image-2.0 具备较强的文本生成能力，但为了保证排版的专业性，需要明确指定字体风格、大小对比和背景融合方式。

**实施步骤**:
1. 指定字体风格，例如“sans-serif bold font”（无衬线粗体）或“elegant serif typography”（优雅的衬线排版）。
2. 描述文本与背景的关系，如“white text on dark blurry background”（深色模糊背景上的白色文字）以确保可读性。
3. 如果是英文内容，建议全部大写以减少生成错误；如果是中文，建议简短有力。
