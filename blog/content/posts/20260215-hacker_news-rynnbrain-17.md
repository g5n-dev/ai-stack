```python
# 示例2：从Hacker News获取标题并分析关键词
import requests
from bs4 import BeautifulSoup

def analyze_hn_titles():
    """
    获取Hacker News首页标题并提取常见关键词
    解决问题：如何抓取网页内容并进行简单的文本分析
    """
    url = "https://news.ycombinator.com/"
    try:
        response = requests.get(url, timeout=10)
        soup = BeautifulSoup(response.text, 'html.parser')
        titles = soup.find_all('span', class_='titleline')
        
        # 简单关键词统计
        keywords = {}
        for title in titles[:10]:  # 只分析前10条
            text = title.get_text().lower()
            for word in text.split():
                if len(word) > 3:  # 只统计长度>3的单词
                    keywords[word] = keywords.get(word, 0) + 1
        
        return sorted(keywords.items(), key=lambda x: x[1], reverse=True)[:5]
    except Exception as e:
        print(f"抓取失败: {e}")
        return []

# 使用示例
top_keywords = analyze_hn_titles()
print("热门关键词:", top_keywords)
```
